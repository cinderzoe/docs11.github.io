BasicTable 组件 ​

基于Elemenet的 Table 组件的二次封装

示例 ​

vue
<template>
  <div class="app-container">
    <BasicTable
      ref="BasicTableRef"
      v-bind="{
        searchSchemas,
        columns: tableColumns,
        dataSource: meetingPageData.records,
        border: true,
        pagination: {
          total: Number(meetingPageData.total),
          page: Number(meetingPageData.current),
          limit: Number(meetingPageData.size),
          pageSize: [10, 20, 50],
        },
        immediate: true,
        actions: tableActions,
      }"
      @change="getMeetingPage"
    >
      <el-col :span="1.5" slot="header">
        <el-button
          v-hasPermi="['sale:meeting:add']"
          @click="toCreateMeeting"
          type="primary"
          plain
          icon="plus-circle"
          size="mini"
        >
          创建
        </el-button>
        <el-button @click="toExport" type="primary" plain icon="plus-circle" size="mini">导出</el-button>
      </el-col>
    </BasicTable>
  </div>
</template>
<script>
import BasicTable from '@/components/BasicTable';

export default {
  name: 'Meeting',
  components: { BasicTable },
  data() {},
  computed: {
    searchSchemas() {
      return [
        {
          label: '会议名称',
          field: 'searchValue',
          component: 'Input',
        },
        {
          label: '会议日期',
          field: 'dateRange',
          component: 'DatePicker',
          componentProps: {
            type: 'daterange',
            valueFormat: 'yyyy-MM-dd',
            rangeSeparator: '至',
            startPlaceholder: '开始日期',
            endPlaceholder: '结束日期',
          },
          valueHandle: (v) => v && `${v[0]},${v[1]}`,
        },
        {
          label: '创建日期',
          field: 'createDateRange',
          component: 'DatePicker',
          componentProps: {
            type: 'daterange',
            valueFormat: 'yyyy-MM-dd',
            rangeSeparator: '至',
            startPlaceholder: '开始日期',
            endPlaceholder: '结束日期',
          },
          valueHandle: (v) => v && `${v[0]},${v[1]}`,
        },
        {
          label: '会议类型',
          field: 'type',
          component: 'Select',
          componentProps: {
            api: meetingTypeApi,
            valueField: 'dictCode',
            labelField: 'dictLabel',
            clearable: true,
          },
        },
        {
          label: '申请部门',
          field: 'deptIds',
          component: 'ApiTree',
          componentProps: {
            api: deptTreeApi,
            labelField: 'name',
            valueField: 'deptId',
          },
        },
        {
          label: '申请人',
          field: 'userIds',
          component: 'ApiTree',
          componentProps: {
            api: deptUserTreeApi,
            valueField: 'userId',
            labelField: 'deptName',
          },
        },
        {
          label: 'OA内部订单号',
          field: 'innerOrder',
          component: 'Input',
          componentProps: {
            placeholder: '请输入',
          },
        },
        {
          label: '会议状态',
          field: 'meetingStatus',
          component: 'Select',
          componentProps: {
            options: [
              { label: MeetingStatusLabelEnum.ToBeSubmit, value: MeetingStatusEnum.ToBeSubmit },
              { label: MeetingStatusLabelEnum.Approving, value: MeetingStatusEnum.Approving },
              { label: MeetingStatusLabelEnum.Approved, value: MeetingStatusEnum.Approved },
              { label: MeetingStatusLabelEnum.SigningUp, value: MeetingStatusEnum.SigningUp },
              { label: MeetingStatusLabelEnum.Executing, value: MeetingStatusEnum.Executing },
              { label: MeetingStatusLabelEnum.Finished, value: MeetingStatusEnum.Finished },
              { label: MeetingStatusLabelEnum.Canceled, value: MeetingStatusEnum.Canceled },
            ],
            clearable: true,
          },
        },
        {
          field: 'enableDataScope',
          ifShow: false,
          defaultValue: true,
        },
      ];
    },
    tableColumns() {
      return [
        { label: '会议名称', field: 'meetingName' },
        { label: '会议编码', field: 'id' },
        { label: '内部订单号', field: 'oaWorkflowRequestId' },
        { label: '申请部门', field: 'deptName' },
        { label: '申请人', field: 'nickName' },
        { label: '会议开始时间', field: 'beginDate', render: (v) => formatToDateTime(v) },
        { label: '会议结束时间', field: 'endDate', render: (v) => formatToDateTime(v) },
        { label: '会议状态', field: 'meetingStatus.desc' },
        { label: '创建时间', field: 'createTime', render: (v) => formatToDateTime(v) },
        { label: '创建人', field: 'nickName' },
      ];
    },
    tableActions() {
      return [
        {
          label: '编辑',
          onClick: this.toEditMeeting,
          ifShow: ({ status, userId }) => {
            return (
              [0, 5].includes(status?.value) && checkPermi(['sale:meeting:edit']) && userId == this.userInfo.userId
            );
          },
        },
        {
          label: '详情',
          onClick: this.toMeetingDetail,
          ifShow: ({ status }) => status,
        },
      ];
    },
  },
};
</script>

Mehtods ​

通过 ref 获取 BasicTable 组件内部的函数

resetTable
类型：() => void
说明：重置表格

refreshTable
类型：(fields: Recordable) => void;
说明：刷新表格

Events ​

change
类型：(params) => void;
说明：当表格发生变化

selectionChange
类型：(selection) => void;
说明：当选择项发生变化时会触发该事件

Props ​

参数名类型默认值说明
searchSchemasArray<SearchSchemas>-搜索表单配置项
columnsArray<TableColumnsType>-表格列配置
dataSourceArray<Object>-表格数据源
paginationPaginationType{total:0,page:1,limit:10,pageSize:[10,20,50]}分页配置对象
actionsArray<TableActionsType>[]操作按钮配置数组
immediateBooleanfalse是否立即执行搜索
selectionBooleanfalse是否显示选择列

SearchSchemas 跟 BasicForm 组件的 schemas 参数配置是一样的，前往

TableColumnsType
ts
interface TableColumnsType {
  label: String;
  field: String;
  render?: T(args:T) => any;
  component:'DatePicker '| 'Select'| 'Cascader'| 'Input' | 'ApiTree' | 'Picture'
}
PaginationType
ts
interface PaginationType {
  total: Number;
  page: Number;
  limit: Number;
  pageSize: Array<Number>;
}
TableActionsType
ts
interface TableActionsType {
  label: String;
  onClick: Function(scope.row);
  ifShow: Boolean | (scope.row, scope)=>Boolean;
  icon?: String;
}
